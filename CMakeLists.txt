cmake_minimum_required(VERSION 3.16)
project(gam3t3ch_sbc_bench VERSION 0.5.2 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(g3bench-native native/bench.c)
if(MSVC)
  target_compile_options(g3bench-native PRIVATE /O2 /W4)
else()
  target_compile_options(g3bench-native PRIVATE -O2 -Wall -Wextra -Wpedantic)
  target_link_libraries(g3bench-native PRIVATE m)
endif()
option(BUILD_GPU "Build the optional SDL2/OpenGL ES 2 benchmark" ON)
if(BUILD_GPU)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
    pkg_check_modules(GLES2 QUIET IMPORTED_TARGET glesv2)
  endif()
  if(SDL2_FOUND AND GLES2_FOUND)
    add_executable(g3bench-gpu native/gpu.c)
    target_link_libraries(g3bench-gpu PRIVATE PkgConfig::SDL2 PkgConfig::GLES2 m)
    target_compile_options(g3bench-gpu PRIVATE -O2 -Wall -Wextra)
  else()
    message(STATUS "SDL2/GLES2 unavailable: CPU/memory remain available")
  endif()
endif()
